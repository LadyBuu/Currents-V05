<!-- PlaceQueries façade — must load before culture (Block 21),
     religion (Block 20), and settlement (Block 35). Defined as a
     lazy façade so LocationState.getPlaceById resolves at call
     time, not at definition time. -->
<script>
    (function () {
        'use strict';
        window.PlaceQueries = {
            getPlaceById: function (id) {
                if (!window.LocationState
                    || typeof window.LocationState.getPlaceById !== 'function') {
                    return null;
                }
                return window.LocationState.getPlaceById(id);
            },
            getPlaceSummaries: function () {
                return (window.LocationQueries
                    && typeof window.LocationQueries.getPlaceSummaries === 'function')
                    ? window.LocationQueries.getPlaceSummaries()
                    : [];
            },
            getAllPlaces: function () {
                return (typeof window.getPlaces === 'function') ? window.getPlaces() : [];
            }
        };
    })();
</script>

<!-- ============================================================
     BLOCK 22b — RELATIONSHIP FLAT-GLOBAL SHIM
     household-bootstrap.js requires window.createRelationship as
     a cross-domain global. RelationshipCommands exports it on a
     namespace. This shim creates the flat-global aliases the
     household bootstrap expects, without modifying the household
     domain. It is a one-way compatibility layer.
     ============================================================ -->
<script>
    (function () {
        'use strict';
        if (!window.RelationshipCommands) return;
        var RC = window.RelationshipCommands;
        if (typeof RC.createRelationship === 'function'
            && typeof window.createRelationship !== 'function') {
            window.createRelationship = function (a, b, options) {
                return RC.createRelationship(a, b, options);
            };
        }
        if (typeof RC.endRelationship === 'function'
            && typeof window.endRelationship !== 'function') {
            window.endRelationship = function (id, year, reason) {
                return RC.endRelationship(id, year, reason);
            };
        }
        if (typeof RC.updateRelationship === 'function'
            && typeof window.updateRelationship !== 'function') {
            window.updateRelationship = function (id, patch) {
                return RC.updateRelationship(id, patch);
            };
        }
        if (typeof RC.getRelationshipById === 'function'
            && typeof window.getRelationshipById !== 'function') {
            window.getRelationshipById = function (id) {
                return RC.getRelationshipById(id);
            };
        }
    })();
</script>
