<!-- ============================================================
     BLOCK 34b — PlaceQueries FAÇADE
     (Settlement factory, view models, and the three settlement
      modals call window.PlaceQueries.getPlaceById. LocationState
      exposes getPlaceById; this façade wires the name settlement
      expects to the implementation that exists. Loaded after
      location-bootstrap.js, before any settlement module.)
     ============================================================ -->
<script>
    (function () {
        'use strict';
        if (!window.LocationState || typeof window.LocationState.getPlaceById !== 'function') {
            throw new Error('place-queries shim: location-state.js must load first.');
        }
        window.PlaceQueries = {
            getPlaceById: function (id) {
                return window.LocationState.getPlaceById(id);
            },
            getPlaceSummaries: function () {
                return (window.LocationQueries && typeof window.LocationQueries.getPlaceSummaries === 'function')
                    ? window.LocationQueries.getPlaceSummaries()
                    : [];
            }
        };
    })();
</script>
